export type WorkflowStatus = 'pending' | 'active' | 'suspended' | 'completed' | 'failed' | 'cancelled';

/** Module Section 1: Handler logic, state machine step 1 */
export interface StepState1 {
  readonly id: string;
  readonly iteration: number;
  readonly payload: Record<string, unknown>;
  readonly status: WorkflowStatus;
  readonly timestamp: number;
}

export class StepHandler1 {
  private state: StepState1;
  constructor(id: string) {
    this.state = { id, iteration: 1, payload: {}, status: 'pending', timestamp: Date.now() };
  }
  public execute(input: Record<string, unknown>): StepState1 {
    const nextIteration = this.state.iteration + 1;
    const status: WorkflowStatus = nextIteration % 10 === 0 ? 'suspended' : 'active';
    this.state = { ...this.state, iteration: nextIteration, payload: { ...this.state.payload, ...input }, status };
    return this.state;
  }
}
